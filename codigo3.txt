    def generate_atomic_hash(self, atomic_number):
        """
        Genera un hash único ultra-complejo para garantizar 118 patrones únicos.
        """
        # Usar múltiples capas de dispersión para máxima unicidad
        hash_val = 0
        
        # Factor 1: Número atómico con multiplicadores primos únicos
        primes = [7919, 11003, 13007, 17011, 19013, 23017, 29021]
        hash_val += atomic_number * primes[atomic_number % len(primes)]
        
        # Factor 2: Propiedades de tabla periódica
        period = self.get_period(atomic_number)
        group = self.get_group(atomic_number)
        hash_val += (period * 31031) + (group * 37037)
        
        # Factor 3: Tipo de elemento con dispersión extrema
        element_type = self.get_element_type(atomic_number)
        hash_val += element_type * 41041
        
        # Factor 4: Análisis de dígitos complejo
        digits = [int(d) for d in str(atomic_number)]
        digits_sum = sum(digits)
        digits_product = 1
        for d in digits:
            digits_product *= (d + 1)
        hash_val += digits_sum * 43043 + digits_product * 47047
        
        # Factor 5: Posición y paridad
        hash_val += (atomic_number % 7) * 53053
        hash_val += ((atomic_number - 1) // 5) * 59059
        
        # Factor 6: Configuración electrónica aproximada
        electron_shells = self.get_electron_distribution(atomic_number)
        for i, electrons in enumerate(electron_shells):
            hash_val += electrons * (61061 + i * 1009)
        
        # Factor 7: Propiedades físicas simuladas
        mass_approx = int(atomic_number * 2.4 * 100)  # Masa atómica aproximada
        hash_val += mass_approx * 67067
        
        # Factor 8: Dispersión por bloques químicos
        block = self.get_block(atomic_number)
        hash_val += block * 71071
        
        return hash_val % (2**31 - 1)
    
    def get_grid_size(self, atomic_number):
        """Determina el tamaño de rejilla único para cada elemento."""
        # Usar número atómico para determinar tamaño de rejilla (20-24)
        base_size = 20 + (atomic_number % 5)  # 20, 21, 22, 23, 24
        return base_size
    
    def get_corner_configuration(self, atomic_number):
        """
        Determina la configuración única de esquinas para cada elemento.
        Retorna: (esquinas_presentes, tamaños_esquinas, formas_esquinas)
        """
        np.random.seed(atomic_number * 12345) # Semilla específica para esquinas
        
        # Determinar qué esquinas están presentes (siempre al menos 2)
        corner_configs = [
            [True, True, True],      # Las 3 esquinas
            [True, True, False],     # Solo superior izq y der
            [True, False, True],     # Solo superior izq e inferior izq
            [False, True, True],     # Solo superior der e inferior izq
            [True, True, True],      # Las 3 esquinas (duplicado para mayor probabilidad)
        ]
        
        config_index = atomic_number % len(corner_configs)
        esquinas_presentes = corner_configs[config_index]
        
        # Determinar tamaños de esquinas (5x5, 6x6, 7x7, 8x8)
        tamaños_base = [5, 6, 7, 8]
        tamaños_esquinas = []
        
        for i in range(3):
            if esquinas_presentes[i]:
                # Cada esquina puede tener diferente tamaño
                size_index = (atomic_number + i * 17) % len(tamaños_base)
                tamaños_esquinas.append(tamaños_base[size_index])
            else:
                tamaños_esquinas.append(0)
        
        # Determinar formas de esquinas (completa, parcial, modificada)
        formas_esquinas = []
        for i in range(3):
            if esquinas_presentes[i]:
                forma_type = (atomic_number + i * 23) % 4
                formas_esquinas.append(forma_type)  # 0=completa, 1=parcial1, 2=parcial2, 3=modificada
            else:
                formas_esquinas.append(-1)
        
        return esquinas_presentes, tamaños_esquinas, formas_esquinas
    
    def get_atomic_number_position(self, atomic_number):
        """Determina la posición del número atómico."""
        positions = [
            "top_left",      # Superior izquierda
            "top_right",     # Superior derecha  
            "bottom_left",   # Inferior izquierda
            "bottom_right"   # Inferior derecha
        ]
        return positions[atomic_number % len(positions)]
    
    def generate_corner_pattern(self, atomic_number, corner_index, size, form_type):
        """
        Genera un patrón de esquina único con tamaño y forma específicos.
        """
        if size == 0:  # Esquina no presente
            return None
            
        # Usar semilla única para esta esquina específica
        seed = atomic_number * 1000 + corner_index * 100 + size * 10 + form_type
        np.random.seed(seed)
        
        # Crear patrón base
        pattern = [[0 for _ in range(size)] for _ in range(size)]
        
        if form_type == 0:  # Patrón completo estándar
            # Borde exterior
            for i in range(size):
                pattern[0][i] = 1
                pattern[size-1][i] = 1
                pattern[i][0] = 1
                pattern[i][size-1] = 1
            
            # Borde interior si es suficientemente grande
            if size >= 7:
                for i in range(2, size-2):
                    pattern[2][i] = 1
                    pattern[size-3][i] = 1
                    pattern[i][2] = 1
                    pattern[i][size-3] = 1
            
            # Centro único basado en número atómico
            center = size // 2
            if size >= 5:
                pattern[center][center] = (atomic_number + corner_index) % 2
                
        elif form_type == 1:  # Patrón parcial tipo 1
            # Solo mitad superior del patrón estándar
            for i in range(size):
                for j in range(size//2 + 1):
                    if i == 0 or i == size//2 or j == 0 or j == size-1:
                        pattern[j][i] = 1
                        
        elif form_type == 2:  # Patrón parcial tipo 2
            # Patrón diagonal
            for i in range(size):
                pattern[i][i] = 1
                if i < size - 1:
                    pattern[i][i+1] = 1
                if i > 0:
                    pattern[i][i-1] = 1
                    
        else:  # form_type == 3, Patrón modificado único
            # Crear patrón único usando Mersenne Twister
            for i in range(size):
                for j in range(size):
                    cell_prob = (atomic_number + i * 7 + j * 11 + corner_index * 13) % 100
                    if cell_prob > 55:
                        pattern[i][j] = 1
        
        return pattern
    
    def generate_timing_patterns(self, atomic_number, grid_size):
        """Genera patrones de timing únicos."""
        timing_patterns = []
        seed = atomic_number * 7777
        np.random.seed(seed)
        
        # Líneas de timing horizontal y vertical con patrones únicos
        timing_row = grid_size // 2
        timing_col = grid_size // 2
        
        # Patrón horizontal
        for col in range(grid_size):
            if col % 2 == (atomic_number % 2):
                cell_hash = (atomic_number + col * 17) % 100
                if cell_hash > 45:
                    timing_patterns.append((timing_row, col, 1))
        
        # Patrón vertical
        for row in range(grid_size):  
            if row % 3 == (atomic_number % 3):
                cell_hash = (atomic_number + row * 19) % 100
                if cell_hash > 45:
                    timing_patterns.append((row, timing_col, 1))
        
        return timing_patterns
    
    def generate_data_pattern(self, atomic_number, grid_size, occupied_positions):
        """Genera el patrón de datos único para el elemento."""
        data_pattern = []
        seed = atomic_number * 9999
        np.random.seed(seed)
        
        for i in range(grid_size):
            for j in range(grid_size):
                if (i, j) not in occupied_positions:
                    # Crear patrón único basado en múltiples factores
                    cell_hash = (
                        atomic_number * 13 + 
                        i * 17 + 
                        j * 19 + 
                        (i + j) * 23 + 
                        (i * j) * 29
                    ) % 100
                    
                    # Threshold variable por elemento
                    threshold = 45 + (atomic_number % 20)
                    
                    # Modificadores adicionales para mayor unicidad
                    if (i + j) % 4 == atomic_number % 4:
                        threshold += 10
                    if i % 5 == atomic_number % 5:
                        threshold -= 5
                    
                    if cell_hash > threshold:
                        data_pattern.append((i, j, 1))
        
        return data_pattern
    
    def generate_qr_marker(self, symbol, atomic_number, show_symbol=True, show_atomic_number=True):
        """
        Genera un marcador QR único con máxima diferenciación para AR.
        """
        img_size = self.marker_size + 2 * self.border_size
        img = Image.new('RGB', (img_size, img_size), color='white')
        draw = ImageDraw.Draw(img)
        
        # Obtener configuraciones únicas para este elemento
        grid_size = self.get_grid_size(atomic_number)
        cell_size = self.marker_size // grid_size
        
        esquinas_presentes, tamaños_esquinas, formas_esquinas = self.get_corner_configuration(atomic_number)
        atomic_pos = self.get_atomic_number_position(atomic_number)
        
        occupied_positions = set()
        
        # Generar y dibujar esquinas únicas
        corner_positions = [(0, 0), (grid_size-max(tamaños_esquinas), 0), (0, grid_size-max(tamaños_esquinas))]
        
        for corner_idx in range(3):
            if esquinas_presentes[corner_idx]:
                size = tamaños_esquinas[corner_idx]
                form = formas_esquinas[corner_idx]
                
                corner_pattern = self.generate_corner_pattern(atomic_number, corner_idx, size, form)
                corner_x, corner_y = corner_positions[corner_idx]
                
                # Ajustar posición para esquinas de diferente tamaño
                if corner_idx == 1:  # Superior derecha
                    corner_x = grid_size - size
                elif corner_idx == 2:  # Inferior izquierda
                    corner_y = grid_size - size
                
                # Dibujar esquina
                for i in range(size):
                    for j in range(size):
                        if corner_pattern[i][j] == 1:
                            x0 = self.border_size + (corner_x + j) * cell_size
                            y0 = self.border_size + (corner_y + i) * cell_size
                            x1 = x0 + cell_size
                            y1 = y0 + cell_size
                            draw.rectangle([(x0, y0), (x1, y1)], fill='black')
                        
                        # Marcar posición como ocupada
                        occupied_positions.add((corner_y + i, corner_x + j))
        
        # Generar patrones de timing únicos
        timing_patterns = self.generate_timing_patterns(atomic_number, grid_size)
        for row, col, value in timing_patterns:
            if value == 1 and 0 <= row < grid_size and 0 <= col < grid_size:
                x0 = self.border_size + col * cell_size
                y0 = self.border_size + row * cell_size
                x1 = x0 + cell_size
                y1 = y0 + cell_size
                draw.rectangle([(x0, y0), (x1, y1)], fill='black')
                occupied_positions.add((row, col))
        
        # Generar patrón de datos único
        data_patterns = self.generate_data_pattern(atomic_number, grid_size, occupied_positions)
        for row, col, value in data_patterns:
            if value == 1:
                x0 = self.border_size + col * cell_size
                y0 = self.border_size + row * cell_size
                x1 = x0 + cell_size
                y1 = y0 + cell_size
                draw.rectangle([(x0, y0), (x1, y1)], fill='black')
                occupied_positions.add((row, col))
        
        # Añadir símbolo químico en el centro
        if show_symbol:
            try:
                font_path = "font/OpenSans-Bold.ttf"
                if len(symbol) == 1:
                    font_size = max(60, cell_size * 3)
                elif len(symbol) == 2:
                    font_size = max(50, cell_size * 2)
                else:
                    font_size = max(40, cell_size * 2)
                font_symbol = ImageFont.truetype(font_path, font_size)
            except IOError:
                try:
                    font_path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
                    font_symbol = ImageFont.truetype(font_path, font_size)
                except IOError:
                    font_symbol = ImageFont.load_default()
            
            # Área central para el símbolo
            symbol_area_size = min(6, grid_size // 3)
            symbol_area_x0 = self.border_size + (grid_size - symbol_area_size) // 2 * cell_size
            symbol_area_y0 = self.border_size + (grid_size - symbol_area_size) // 2 * cell_size
            symbol_area_x1 = symbol_area_x0 + cell_size * symbol_area_size
            symbol_area_y1 = symbol_area_y0 + cell_size * symbol_area_size
            
            # Fondo blanco para el símbolo
            draw.rectangle(
                [(symbol_area_x0, symbol_area_y0), (symbol_area_x1, symbol_area_y1)],
                fill='white'
            )
            
            # Centrar texto
            symbol_width = draw.textlength(symbol, font=font_symbol)
            bbox = draw.textbbox((0, 0), symbol, font=font_symbol)
            text_height = bbox[3] - bbox[1]
            
            text_x = symbol_area_x0 + (symbol_area_size * cell_size - symbol_width) / 2
            text_y = symbol_area_y0 + (symbol_area_size * cell_size - text_height) / 2 - bbox[1]
            
            draw.text((text_x, text_y), symbol, fill='black', font=font_symbol)
        
        # Añadir número atómico en posición única
        if show_atomic_number:
            try:
                font_path = "font/OpenSans-Bold.ttf"
                atomic_text = str(atomic_number)
                font_size = max(40, cell_size * 2)
                if len(atomic_text) >= 3:
                    font_size = max(35, cell_size * 1.5)
                font_info = ImageFont.truetype(font_path, font_size)
            except IOError:
                try:
                    font_path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
                    font_info = ImageFont.truetype(font_path, font_size)
                except IOError:
                    font_info = ImageFont.load_default()
            
            # Determinar posición según configuración
            area_size = 4
            if atomic_pos == "top_left":
                atomic_x0 = self.border_size
                atomic_y0 = self.border_size
            elif atomic_pos == "top_right":
                atomic_x0 = self.border_size + (grid_size - area_size) * cell_size
                atomic_y0 = self.border_size
            elif atomic_pos == "bottom_left":
                atomic_x0 = self.border_size
                atomic_y0 = self.border_size + (grid_size - area_size) * cell_size
            else:  # bottom_right
                atomic_x0 = self.border_size + (grid_size - area_size) * cell_size
                atomic_y0 = self.border_size + (grid_size - area_size) * cell_size
            
            atomic_x1 = atomic_x0 + cell_size * area_size
            atomic_y1 = atomic_y0 + cell_size * area_size
            
            # Fondo blanco
            draw.rectangle([(atomic_x0, atomic_y0), (atomic_x1, atomic_y1)], fill='white')
            
            # Centrar texto
            text_width = draw.textlength(atomic_text, font=font_info)
            bbox = draw.textbbox((0, 0), atomic_text, font=font_info)
            text_height = bbox[3] - bbox[1]
            
            text_x = atomic_x0 + (area_size * cell_size - text_width) / 2
            text_y = atomic_y0 + (area_size * cell_size - text_height) / 2 - bbox[1]
            
            draw.text((text_x, text_y), atomic_text, fill='black', font=font_info)
        
        marker_array = np.array(img)
        marker_gray = cv2.cvtColor(marker_array, cv2.COLOR_RGB2GRAY)
        
        return marker_gray
    
    def generate_element_marker(self, symbol, name, atomic_number, show_symbol=False, show_atomic_number=True, symbol_size=2):
        """
        Genera un marcador RA tradicional único mejorado.
        """
        img_size = self.marker_size + 2 * self.border_size
        img = Image.new('RGB', (img_size, img_size), color='white')
        draw = ImageDraw.Draw(img)
        
        # Usar hash atómico único
        unique_hash = self.generate_atomic_hash(atomic_number)
        np.random.seed(unique_hash)
        
        grid_size = 12 + (atomic_number % 4)  # 12, 13, 14, 15
        cell_size = self.marker_size // grid_size
        
        for i in range(grid_size):
            for j in range(grid_size):
                if show_symbol and i < symbol_size * 2 and j < symbol_size * 2:
                    continue
                if show_atomic_number and i >= grid_size - 3 and j >= grid_size - 3:
                    continue
                
                # Patrón único ultra-complejo
                cell_hash = (
                    unique_hash + 
                    i * 73 + 
                    j * 79 + 
                    (i + j) * 83 + 
                    (i * j) * 89
                ) % 100
                
                threshold = 50 + ((unique_hash + i + j) % 5) * 4
                
                if cell_hash > threshold:
                    x0 = self.border_size + j * cell_size
                    y0 = self.border_size + i * cell_size
                    x1 = x0 + cell_size
                    y1 = y0 + cell_size
                    draw.rectangle([(x0, y0), (x1, y1)], fill='black')
        
        # Añadir símbolo y número atómico (código similar al QR pero adaptado)
        if show_symbol:
            try:
                font_path = "font/OpenSans-Bold.ttf"
                font_size = cell_size * symbol_size * 2
                font_symbol = ImageFont.truetype(font_path, font_size)
            except IOError:
                font_symbol = ImageFont.load_default()
            
            symbol_area_x0 = self.border_size
            symbol_area_y0 = self.border_size
            symbol_area_x1 = symbol_area_x0 + cell_size * symbol_size * 2
            symbol_area_y1 = symbol_area_y0 + cell_size * symbol_size * 2
            
            draw.rectangle([(symbol_area_x0, symbol_area_y0), (symbol_area_x1, symbol_area_y1)], fill='white')
            
            symbol_width = draw.textlength(symbol, font=font_symbol)
            text_x = symbol_area_x0 + (symbol_size * 2 * cell_size - symbol_width) / 2
            text_y = symbol_area_y0 + (symbol_size * 2 * cell_size - font_size) / 2
            
            draw.text((text_x, text_y), symbol, fill='black', font=font_symbol)
        
        if show_atomic_number:
            atomic_text = str(atomic_number)
            try:
                font_path = "font/OpenSans-Bold.ttf"
                font_size = cell_size * 2
                font_info = ImageFont.truetype(font_path, font_size)
            except IOError:
                font_info = ImageFont.load_default()
            
            atomic_area_x0 = self.border_size + (grid_size - 3) * cell_size
            atomic_area_y0 = self.border_size + (grid_size - 3) * cell_size
            atomic_area_x1 = atomic_area_x0 + cell_size * 3
            atomic_area_y1 = atomic_area_y0 + cell_size * 3
            
            draw.rectangle([(atomic_area_x0, atomic_area_y0), (atomic_area_x1, atomic_area_y1)], fill='white')
            
            text_width = draw.textlength(atomic_text, font=font_info)
            text_x = atomic_area_x0 + (3 * cell_size - text_width) / 2
            text_y = atomic_area_y0 + (3 * cell_size - font_size) / 2
            
            draw.text((text_x, text_y), atomic_text, fill='black', font=font_info)
        
        marker_array = np.array(img)
        marker_gray = cv2.cvtColor(marker_array, cv2.COLOR_RGB2GRAY)
        
        return marker_gray

    # Métodos auxiliares para cálculos químicos
    def get_period(self, atomic_number):
        if atomic_number <= 2: return 1
        elif atomic_number <= 10: return 2
        elif atomic_number <= 18: return 3
        elif atomic_number <= 36: return 4
        elif atomic_number <= 54: return 5
        elif atomic_number <= 86: return 6
        else: return 7
    
    def get_group(self, atomic_number):
        noble_gases = [2, 10, 18, 36, 54, 86, 118]
        for i, ng in enumerate(noble_gases):
            if atomic_number <= ng:
                return (atomic_number - (noble_gases[i-1] if i > 0 else 0)) % 18 + 1
        return 18
    
    def get_element_type(self, atomic_number):
        if atomic_number in [3, 11, 19, 37, 55, 87]: return 1
        elif atomic_number in [4, 12, 20, 38, 56, 88]: return 2
        elif atomic_number in list(range(21, 31)) + list(range(39, 49)) + list(range(72, 81)) + list(range(104, 113)): return 3
        elif atomic_number in [5, 14, 32, 33, 51, 52, 85]: return 4
        elif atomic_number in [2, 10, 18, 36, 54, 86, 118]: return 5
        elif 57 <= atomic_number <= 71: return 6
        elif 89 <= atomic_number <= 103: return 7
        else: return 8
    
    def get_electron_distribution(self, atomic_number):
        shells = [2, 8, 18, 32, 32, 18, 8]
        distribution = []
        remaining = atomic_number
        
        for capacity in shells:
            if remaining <= 0:
                distribution.append(0)
            else:
                electrons = min(remaining, capacity)
                distribution.append(electrons)
                remaining -= electrons
        
        return distribution
    
    def get_block(self, atomic_number):
        if atomic_number in [1, 2]: return 1  # s-block
        elif atomic_number in list(range(3, 13)) + list(range(19, 21)) + list(range(37, 39)) + list(range(55, 57)) + list(range(87, 89)): return 1  # s-block
        elif atomic_number in list(range(13, 19)) + list(range(31, 37)) + list(range(49, 55)) + list(range(81, 87)) + list(range(113, 119)): return 2  # p-block
        elif atomic_number in list(range(21, 31)) + list(range(39, 49)) + list(range(72, 81)) + list(range(104, 113)): return 3  # d-block
        else: return 4  # f-block

    def get_element_by_atomic_number(self, atomic_number):
        """Obtiene un elemento por su número atómico."""
        for element in self.elements:
            if element[2] == atomic_number:
                return element
        return None

def get_image_download_link(img, filename, text):
    """Genera un enlace para descargar una imagen."""
    buffered = io.BytesIO()
    img.save(buffered, format="PNG")
    img_str = base64.b64encode(buffered.getvalue()).decode()
    href = f'<a href="data:file/png;base64,{img_str}" download="{filename}">{text}</a>'
    return href
